"""Task 9 execution guard, external to frozen product and tests."""
import os
from pathlib import Path
import subprocess
import sys

GUARD = str(Path(__file__).resolve().parent)
BASE = str(Path(sys._base_executable).resolve())

def audit(event, args):
    if event in {'socket.connect', 'socket.getaddrinfo', 'socket.gethostbyname',
                 'socket.gethostbyaddr', 'socket.sendto', 'socket.sendmsg',
                 'os.system', 'os.exec', 'os.posix_spawn'}:
        raise RuntimeError('Task 9 offline guard denied ' + event)

sys.addaudithook(audit)
_original_popen = subprocess.Popen

class GuardedPopen(_original_popen):
    def __init__(self, args, *positional, **kwargs):
        if isinstance(args, (str, bytes)) or kwargs.get('shell'):
            raise RuntimeError('Task 9 offline guard requires an explicit Python argv')
        executable = kwargs.get('executable') or args[0]
        if str(Path(executable).resolve()) != BASE:
            raise RuntimeError('Task 9 offline guard denied non-Python subprocess')
        if any(arg in ('-I', '-E', '-S') or isinstance(arg, str) and arg.startswith('-I') for arg in args[1:]):
            raise RuntimeError('Task 9 offline guard requires inherited Python site initialization')
        env = dict(os.environ if kwargs.get('env') is None else kwargs['env'])
        env['PYTHONPATH'] = GUARD + os.pathsep + env.get('PYTHONPATH', '')
        env['PYTHONDONTWRITEBYTECODE'] = '1'
        kwargs['env'] = env
        super().__init__(args, *positional, **kwargs)

subprocess.Popen = GuardedPopen
